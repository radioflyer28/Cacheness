```python
    def _generate_new_key(self) -> bytes:
        key = secrets.token_bytes(32)
        if self.use_in_memory_key:
            return key
        try:
            self.key_file_path.parent.mkdir(parents=True, exist_ok=True)
            self.key_file_path.write_bytes(key)
            self._set_key_file_permissions(self.key_file_path)
            return key
        except OSError as e:
            if self.raise_on_key_fallback:
                from .error_handling import CacheSecurityError
                raise CacheSecurityError(...) from e
            logger.error(f"Failed to generate signing key: {e}")
            logger.warning("Using in-memory signing key (not persistent)")
            return key
```

From src/cacheness/security.py (_load_or_generate_key, lines 109-131):
```python
    def _load_or_generate_key(self) -> bytes:
        if self.use_in_memory_key:
            return secrets.token_bytes(32)
        try:
            if self.key_file_path.exists():
                key = self.key_file_path.read_bytes()
                if len(key) != 32:
                    logger.warning(f"Invalid key length ({len(key)} bytes), generating new key")
                    return self._generate_new_key()
                return key
            else:
                return self._generate_new_key()
        except (OSError, ValueError) as e:
            logger.warning(f"Failed to load signing key: {e}, generating new key")
            return self._generate_new_key()
```

    ```python
    class TestKeyFallbackPolicyModes:
        """Test each key_fallback_policy mode with write failures."""

        def test_raise_mode_raises_on_write_failure(self, tmp_path):
            """key_fallback_policy='raise' raises CacheSecurityError on key write failure."""
            blocker = tmp_path / "blocker"
            blocker.write_text("block")
    ```
